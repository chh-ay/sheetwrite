```ts prelude="core" title="dashboards.ts"
// A dashboard sheet lays out pivots, charts, KPI cards and slicers on twelve columns
grid.addSheet({ id: "dashboard", name: "Sales dashboard", kind: "dashboard" });
grid.setPivot(
  {
    id: "by-region",
    name: "Revenue by region",
    source: { table: "orders" },
    anchor: { row: 0, col: 0 },
    rows: [1],
    columns: [],
    values: [{ field: 4, aggregation: "SUM", name: "Revenue" }],
  },
  "dashboard",
);

const items: DashboardItem[] = [
  // A KPI card shows a data model measure
  { id: "revenue-kpi", kind: "kpi", measure: "revenue", box: { x: 0, y: 0, w: 3, h: 3 } },
  { id: "region-pivot", kind: "pivot", pivot: "by-region", box: { x: 0, y: 3, w: 6, h: 6 } },
  // The slicer filters the KPI card and the pivot together
  {
    id: "region-slicer",
    kind: "slicer",
    field: { table: "orders", column: "region" },
    selected: ["EMEA"],
    box: { x: 6, y: 0, w: 3, h: 3 },
  },
];

// Each item change is one undo step, saved and synced with the workbook
grid.applyTransaction({
  patches: items.map((item) => ({ op: "setDashboardItem", sheet: "dashboard", item })),
});
```
