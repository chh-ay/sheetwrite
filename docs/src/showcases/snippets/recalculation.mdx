```ts prelude="core" title="recalculation.ts"
// 100,000 revenue formulas that all read H1, loaded as columnar data
const revenue = Array.from({ length: 100_000 }, (_, row) => ({
  kind: "formula" as const,
  src: `=B${row + 1}*C${row + 1}*(1+$H$1)`,
}));
const grid = createGrid(host, { workbook, data: { rowCount: 100_000, columns: { revenue } } });

// One edit, timed until a dependent total is readable, then until the next frame
const started = performance.now();
grid.applyTransaction({
  patches: [{ op: "set", addr: { sheet: "model", row: 0, col: 7 }, value: { kind: "literal", value: 0.05 } }],
});
grid.store.getCell({ sheet: "model", row: 1, col: 7 }).resolved;
const recalcMs = performance.now() - started;
requestAnimationFrame(() => requestAnimationFrame(() => console.info(recalcMs, performance.now() - started)));
```
