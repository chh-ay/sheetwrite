```ts prelude="none" title="formula-assist.ts"
import { createGrid, initSheetwrite } from "@sheetwrite/core";
import { createFormulaBar } from "@sheetwrite/core/shell";
import { formulaAssistAddon } from "@sheetwrite/formula-assist";

// Install once per page; grids created afterwards get suggestions in their formula bar.
await initSheetwrite({ addons: [formulaAssistAddon()] });

const gridHost = document.getElementById("grid");
const barHost = document.getElementById("formula-bar");
if (gridHost === null || barHost === null) throw new Error("The page needs #grid and #formula-bar");

const grid = createGrid(gridHost, {
  workbook: {
    activeSheet: "budget",
    sheets: [{ id: "budget", name: "Budget", rowCount: 20, columns: [] }],
  },
});
createFormulaBar(barHost, grid);

// In the formula bar: type "=" under a column of =SUM(B2:G2), =SUM(B3:G3), …
// Tab takes the ghost text, Ctrl+→ one word, Esc hides it until you type.
```
