```ts
import { createGrid, initSheetwrite, type Sheet } from "@sheetwrite/core";
import "@sheetwrite/core/styles.css";

await initSheetwrite();

const sheet: Sheet = {
  id: "sales",
  name: "Sales",
  rowCount: 3,
  columns: [
    { key: "product", header: "Product", width: 180, type: "text" },
    { key: "amount", header: "Amount", width: 100, type: "number" },
  ],
};

const grid = createGrid(host, {
  workbook: { activeSheet: "sales", sheets: [sheet] },
  presentation: "data-grid",
  data: {
    rowCount: 3,
    columns: { product: ["Notebook", "Pen", "Folder"], amount: [12, 4, 9] },
  },
});
```
